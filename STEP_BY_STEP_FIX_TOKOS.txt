-- CREATE MISSING TOKOS TABLE
CREATE TABLE IF NOT EXISTS tokos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nama TEXT NOT NULL,
    zona_id UUID NOT NULL REFERENCES zonas(id) ON DELETE CASCADE,
    kota TEXT,
    provinsi TEXT,
    alamat TEXT,
    contact_person TEXT,
    phone TEXT,
    email TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    deleted_at TIMESTAMP WITH TIME ZONE,
    UNIQUE(nama, zona_id)
);

-- CREATE INDEXES
CREATE INDEX IF NOT EXISTS idx_tokos_zona_id ON tokos(zona_id);
CREATE INDEX IF NOT EXISTS idx_tokos_deleted_at ON tokos(deleted_at) WHERE deleted_at IS NULL;
CREATE INDEX IF NOT EXISTS idx_tokos_nama ON tokos(nama);

-- GRANT PERMISSIONS
GRANT ALL ON tokos TO authenticated;
GRANT ALL ON tokos TO service_role;

-- SEED SAMPLE DATA
INSERT INTO tokos (nama, zona_id, kota, provinsi)
SELECT 
    'Toko ' || z.nama as nama,
    z.id,
    CASE 
        WHEN z.nama LIKE '%Karawang%' THEN 'Karawang'
        WHEN z.nama LIKE '%Jakarta%' THEN 'Jakarta'
        ELSE 'Indonesia'
    END,
    'Indonesia'
FROM zonas z
ON CONFLICT (nama, zona_id) DO NOTHING;

-- VERIFY
SELECT 'tokos table' as table_name, COUNT(*) as record_count FROM tokos;
